# 이 사이트는 검색에 노출하지 않는다.
#
# 그런데 여기서 전부 Disallow 하지는 않는다 — 크롤러가 페이지를 아예 못
# 가져가면 응답에 실린 noindex(X-Robots-Tag 헤더와 index.html 의 meta
# robots)를 읽을 기회도 없어지고, 그러면 다른 곳에 걸린 링크만 보고 주소를
# 색인에 남기는 일이 오히려 생긴다. 가져가게 두고, 가져간 응답이 "색인하지
# 말라"고 말하게 한다.
#
# 그리고 이 앱의 링크는 대부분 카카오톡 채팅방에 붙는다(PRODUCT.md §투표자).
# 채팅방의 카드가 뜨려면 카카오 스크래퍼가 HTML 을 가져갈 수 있어야 하는데,
# 그 스크래퍼는 색인기가 아니라 미리보기를 만드는 것이라 noindex 와 부딪히지
# 않는다 — 검색에는 안 뜨고 채팅방 카드는 뜬다. robots.txt 로 막는 것만이
# 카드를 죽인다. 그래서 아래에 명시적으로 열어 둔다.

# 카카오톡 링크 카드. UA 는
# `facebookexternalhit/1.1;kakaotalk-scrap/1.0; (+https://devtalk.kakao.com/t/scrap/33984)`
# 이라 두 이름 다 적어 둔다.
User-agent: kakaotalk-scrap
Allow: /

User-agent: facebookexternalhit
Allow: /

# 같은 이유로 나머지 미리보기 스크래퍼들(트위터·슬랙·디스코드 등)도 막지
# 않는다. 아래 규칙 하나가 그것까지 함께 덮는다.
User-agent: *
Allow: /

# 관리자 화면은 Cloudflare Access 뒤라 크롤러가 애초에 들어올 수 없지만,
# 주소가 로그·리퍼러로 새는 것까지 굳이 도울 이유는 없다.
Disallow: /admin
Disallow: /api/
